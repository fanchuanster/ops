# What the deployed service needs — the handoff poller and the job API.
#
# Everything except the local OCR engine. Since OCR moved to Google
# Document AI, which the web application calls, a book arriving through
# the portal arrives as *text*: either the OCR handoff document or an
# upload that was already text. Nothing in that path rasterises a page or
# runs a model.
#
# That matters because paddlepaddle and its models are the better part of
# a gigabyte, and an image carrying them to run a poller is a gigabyte
# pulled, stored and rebuilt for nothing.
#
# `requirements.txt` includes this file and adds the OCR engine, so the
# CLI — which does still run OCR locally — installs unchanged.

# PDF parsing and page rasterization. Ships its own MuPDF, so no poppler
# or system libraries are needed.
pymupdf>=1.28

# The editable master.
python-docx>=1.2

# Reads page dimensions directly in the OCR backend, and pulled in by
# several of the below regardless.
pillow>=10

# The AI correction stage talks to an OpenAI-compatible endpoint over
# plain HTTP; no provider SDK is needed for one endpoint.
httpx>=0.28

# Reads XAI_API_KEY from the repo-root .env for local CLI runs. In a
# container the environment is set directly and no .env exists.
python-dotenv>=1.0

# The reader-facing formats. ebooklib writes valid EPUB 3; WeasyPrint
# renders the PDF variants from the same HTML, with no browser and no
# display needed — which is what makes this service deployable somewhere
# small. See app/pdf/builder.py for why not Playwright.
ebooklib>=0.18
weasyprint>=62

# The asynchronous job API (CLAUDE.md section 13). A scanned book takes
# hours, so the request must not wait for it.
fastapi>=0.115
uvicorn[standard]>=0.32
pydantic>=2.9

# R2 over the S3 API. This service is not a Worker and so has no
# binding; it is the one component that legitimately holds a key.
boto3>=1.35
